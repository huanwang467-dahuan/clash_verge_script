function main(config) {
  // ====================
  // 1. 基础性能配置
  // ====================
  config['log-level'] = 'warn';
  config['allow-lan'] = true;
  config['udp-relay'] = true;
  config['ipv6'] = false;

  // ====================
  // 2. 策略组配置
  // ====================
  const mainProxyGroup = 
    config['proxy-groups'] && config['proxy-groups'][0] 
      ? config['proxy-groups'][0].name 
      : 'Proxy';

  if (Array.isArray(config['proxy-groups'])) {
    config['proxy-groups'].forEach((group) => {
      if (group.type === 'url-test' || group.type === 'least-latency') {
        group['url'] = 'http://cp.cloudflare.com/generate_204';
        group['interval'] = 180;
      }
    });
  }

  // ====================
  // 3. 核心规则配置
  // ====================
  const unityRules = [
    `DOMAIN-SUFFIX,unity.com,${mainProxyGroup}`,
    `DOMAIN-SUFFIX,unity3d.com,${mainProxyGroup}`,
    `DOMAIN-SUFFIX,packages.unity.com,${mainProxyGroup}`
  ];

  const globalRules = [
    `GEOSITE,geolocation-!cn,${mainProxyGroup}`,
    `GEOIP,!(CN),${mainProxyGroup}`,
    `GEOSITE,cn,DIRECT`,
    `GEOIP,CN,DIRECT`,
    `GEOIP,PRIVATE,DIRECT`
  ];

  if (!config['rules']) config['rules'] = [];
  config['rules'] = [...unityRules, ...globalRules, ...config['rules']];

  return config;
}
